get_mean <- function(post){
  output <- 
    post %>% 
    select(starts_with("r_")) %>% 
    pivot_longer(everything()) %>% 
    group_by(name) %>% 
    mean_hdci(value) %>% 
    mutate(tank = as.numeric(str_extract(name, "\\d+"))) %>% 
    mutate(prop = inv_logit_scaled(value))
  
  return(output)
}
