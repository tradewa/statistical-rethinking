  if (mean <= 0 | mean >= 1) stop("must have 0 < mean < 1")
  if (kappa <= 0) stop("kappa must be > 0")
  a <- mean * kappa
  b <- (1.0 - mean) * kappa
  return(list(a = a, b = b))

dordlogit <- function(x, phi, a, log = FALSE) {
  a <- c(as.numeric(a), Inf)
  p <- logistic(a[x] - phi)
  na <- c(-Inf, a)
  np <- logistic(na[x] - phi)
  p <- p - np
  if(log == TRUE) p <- log(p)
  p
}
